private const val MIN_TEMP = 16
private const val MAX_TEMP = 30
class MainActivity  ComponentActivity() {
    override fun onCreate(savedInstanceState Bundle) {
        super.onCreate(savedInstanceState)
        setContent {
            TemperatureScreen()
        }
    }
}
@Composable
fun TemperatureScreen() {
    var temperature by remember { mutableIntStateOf(22) }

    val statusText = when {
        temperature  25 - Жарко
        temperature  20 - Прохладно
        else - Комфортная температура
    }

    val statusColor = when {
        temperature  25 - Color.Red
        temperature  20 - Color.Blue
        else - Color.Cyan
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(24.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        Text(
            text = Температура,
            fontSize = 22.sp,
            fontWeight = FontWeight.Bold
        )

        Spacer(modifier = Modifier.height(24.dp))

        Text(
            text = $temperature °C,
            fontSize = 48.sp,
            fontWeight = FontWeight.Bold
        )

        Spacer(modifier = Modifier.height(24.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            Button(
                onClick = { if (temperature  MIN_TEMP) temperature-- },
                enabled = temperature  MIN_TEMP,
                modifier = Modifier.weight(1f)
            ) {
                Text(−, fontSize = 24.sp)
            }

            Button(
                onClick = { if (temperature  MAX_TEMP) temperature++ },
                enabled = temperature  MAX_TEMP,
                modifier = Modifier.weight(1f)
            ) {
                Text(+, fontSize = 24.sp)
            }
        }

        Spacer(modifier = Modifier.height(24.dp))

        Card(
            modifier = Modifier.fillMaxWidth(),
            shape = RoundedCornerShape(16.dp),
            colors = CardDefaults.cardColors(containerColor = statusColor)
        ) {
            Text(
                text = statusText,
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(16.dp),
                textAlign = TextAlign.Center,
                fontSize = 16.sp,
                fontWeight = FontWeight.Medium
            )
        }
    }
}
